import numpy as np


#------------------------------ TASK1(a) ---------------------------------------
# In order to construct our matrices, we start first by creating a list of all
# the nodes that we have and counting them. For the edges, we create a list of 
# tuples, each one of those has the two nodes of the edge, and since we consider
# them undirected, we only need to write the edge one way. After that we create
# a squared matrix of zeroes with the total number of nodes as rows and columns.
# For filling the adjacency matrix we loop on each tuple in our edges list, take
# the id of the nodes and fill the corresponding line and row with 1. On the other
# hand, we only need the sum of each line for the degree matrix. Finally we print
# # the two matrices. 
#-------------------------------------------------------------------------------
nodes = ['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I']
num_nodes = len(nodes)
edges = [
    ('A', 'B'), ('A', 'C'), ('A', 'D'), ('A', 'H'), ('A', 'I'),
    ('B', 'C'), ('B', 'H'),
    ('C', 'D'), ('C', 'E'),
    ('D', 'G'),
    ('E', 'F'),
    ('F', 'G'), ('F', 'H'),
    ('G', 'H'),
    ('H', 'I')
]
A = np.zeros((num_nodes, num_nodes), dtype=int)
for u, v in edges:
    i, j = nodes.index(u), nodes.index(v)
    A[i, j] = 1
    A[j, i] = 1  
degrees = np.sum(A, axis=1)
D = np.diag(degrees)
print("Adjacency Matrix (A):")
print(A)
print("\nDegree Matrix (D):")
print(D)
#-------------------------------------------------------------------------------


#------------------------------ TASK1(c) ---------------------------------------
# To create the Laplacian matrix, we need to substract the adjacency matrix from
# the degree matrix. Then, we get the eigenvalues and eigenvectors using the "eigh"
# function from the linear algebra functions of numpy, then we round the given 
# values to 4 decimals for a clean visual. And finally we print the result.
#-------------------------------------------------------------------------------
L = D - A
eigenvalues, eigenvectors = np.linalg.eigh(L)
eigenvalues_clean = np.round(eigenvalues, 4)
eigenvectors_clean = np.round(eigenvectors, 4)
print("--- Task 1(c) Output ---")
print("\nLaplacian Matrix (L):")
print(L)
print("\nEigenvalues (Sorted):")
print(eigenvalues_clean)
print("\nEigenvectors:")
print(eigenvectors_clean)
